from flask import Flask, render_template, request, redirect, url_for,
flash, send_file import sqlite3, os, csv, io from datetime import
datetime

app = Flask(name) app.secret_key = “email-marketing-capstone” DB_PATH =

def get_db(): conn = sqlite3.connect(DB_PATH) conn.row_factory =
sqlite3.Row return conn

def init_db(): conn = get_db() conn.executescript(““” CREATE TABLE IF
NOT EXISTS subscribers( id INTEGER PRIMARY KEY AUTOINCREMENT, name TEXT
NOT NULL, email TEXT NOT NULL UNIQUE, customer_group TEXT DEFAULT
‘General’, status TEXT DEFAULT ‘Active’, created_at TEXT NOT NULL );

    CREATE TABLE IF NOT EXISTS campaigns(
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        name TEXT NOT NULL,
        subject TEXT NOT NULL,
        sender TEXT NOT NULL,
        audience TEXT NOT NULL,
        template_name TEXT NOT NULL,
        body TEXT NOT NULL,
        scheduled_at TEXT,
        status TEXT DEFAULT 'Draft',
        campaign_type TEXT DEFAULT 'Promotional',
        created_at TEXT NOT NULL
    );

    CREATE TABLE IF NOT EXISTS analytics(
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        campaign_id INTEGER NOT NULL,
        delivered INTEGER DEFAULT 0,
        opened INTEGER DEFAULT 0,
        clicked INTEGER DEFAULT 0,
        recorded_at TEXT NOT NULL,
        FOREIGN KEY(campaign_id) REFERENCES campaigns(id)
    );
    """)
    conn.commit()
    conn.close()

@app.route(“/”) def dashboard(): conn = get_db() subscriber_count =
conn.execute(“SELECT COUNT() c FROM subscribers”).fetchone()[”c”]
campaign_count = conn.execute(”SELECT COUNT() c FROM
campaigns”).fetchone()[“c”] sent_count = conn.execute(“SELECT COUNT(*) c
FROM campaigns WHERE status=‘Sent’“).fetchone()[“c”]

    totals = conn.execute("""
        SELECT COALESCE(SUM(delivered),0) delivered,
               COALESCE(SUM(opened),0) opened,
               COALESCE(SUM(clicked),0) clicked
        FROM analytics
    """).fetchone()

    open_rate = round((totals["opened"]/totals["delivered"]*100),1) if totals["delivered"] else 0
    click_rate = round((totals["clicked"]/totals["delivered"]*100),1) if totals["delivered"] else 0

    recent = conn.execute("""
        SELECT c.*, a.delivered, a.opened, a.clicked
        FROM campaigns c
        LEFT JOIN analytics a ON a.campaign_id=c.id
        ORDER BY c.id DESC LIMIT 5
    """).fetchall()

    conn.close()
    return render_template("dashboard.html",
                           subscriber_count=subscriber_count,
                           campaign_count=campaign_count,
                           sent_count=sent_count,
                           open_rate=open_rate,
                           click_rate=click_rate,
                           recent=recent)

@app.route(“/subscribers/add”, methods=[“POST”]) def add_subscriber():
name = request.form.get(“name”,““).strip() email =
request.form.get(”email”,““).strip() customer_group =
request.form.get(”customer_group”,“General”).strip() or “General” status
= request.form.get(“status”,“Active”) if not name or not email:
flash(“Name and email are required.”, “danger”) return
redirect(url_for(“subscribers”))

    try:
        conn = get_db()
        conn.execute("""
            INSERT INTO subscribers(name,email,customer_group,status,created_at)
            VALUES(?,?,?,?,?)
        """,(name,email,customer_group,status,datetime.now().isoformat(timespec="seconds")))
        conn.commit()
        conn.close()
        flash("Subscriber added successfully.", "success")
    except sqlite3.IntegrityError:
        flash("This email already exists.", "warning")

    return redirect(url_for("subscribers"))

@app.route(“/subscribers/import”, methods=[“POST”]) def
import_subscribers(): file = request.files.get(“file”) if not file:
flash(“Please choose a CSV file.”, “warning”) return
redirect(url_for(“subscribers”))

    content = file.stream.read().decode("utf-8-sig")
    reader = csv.DictReader(io.StringIO(content))
    conn = get_db()
    added = 0

    for row in reader:
        name = (row.get("name") or "").strip()
        email = (row.get("email") or "").strip()
        group = (row.get("customer_group") or "General").strip()
        if not name or not email:
            continue
        try:
            conn.execute("""
                INSERT INTO subscribers(name,email,customer_group,status,created_at)
                VALUES(?,?,?,?,?)
            """,(name,email,group,"Active",datetime.now().isoformat(timespec="seconds")))
            added += 1
        except sqlite3.IntegrityError:
            pass

    conn.commit()
    conn.close()
    flash(f"{added} subscriber(s) imported.", "success")
    return redirect(url_for("subscribers"))

@app.route(“/campaigns//send”, methods=[“POST”]) def send_campaign(cid):
conn = get_db() campaign = conn.execute(“SELECT * FROM campaigns WHERE
id=?”,(cid,)).fetchone()

    if not campaign:
        conn.close()
        flash("Campaign not found.", "danger")
        return redirect(url_for("campaigns"))

    if campaign["audience"] == "All":
        delivered = conn.execute("SELECT COUNT(*) c FROM subscribers WHERE status='Active'").fetchone()["c"]
    else:
        delivered = conn.execute("""
            SELECT COUNT(*) c FROM subscribers
            WHERE status='Active' AND customer_group=?
        """,(campaign["audience"],)).fetchone()["c"]

    # Safe academic simulation
    opened = int(delivered * 0.62)
    clicked = int(delivered * 0.21)

    conn.execute("UPDATE campaigns SET status='Sent' WHERE id=?",(cid,))
    conn.execute("DELETE FROM analytics WHERE campaign_id=?",(cid,))
    conn.execute("""
        INSERT INTO analytics(campaign_id,delivered,opened,clicked,recorded_at)
        VALUES(?,?,?,?,?)
    """,(cid,delivered,opened,clicked,datetime.now().isoformat(timespec="seconds")))

    conn.commit()
    conn.close()
    flash("Campaign processed successfully and analytics recorded.", "success")
    return redirect(url_for("campaigns"))

COALESCE(a.clicked,0) clicked FROM campaigns c LEFT JOIN analytics a ON

return render_template(”reports.html”, rows=rows)

@app.route(“/reports/export”) def export_report(): conn = get_db() rows
= conn.execute(““” SELECT
c.name,c.subject,c.audience,c.campaign_type,c.status,
COALESCE(a.delivered,0) delivered, COALESCE(a.opened,0) opened,
COALESCE(a.clicked,0) clicked FROM campaigns c LEFT JOIN analytics a ON
c.id=a.campaign_id ORDER BY c.id DESC “““).fetchall() conn.close()

    output = io.StringIO()
    writer = csv.writer(output)
    writer.writerow(["Campaign","Subject","Audience","Type","Status","Delivered","Opened","Clicked","Open Rate %","Click Rate %"])

    for r in rows:
        open_rate = round(r["opened"]/r["delivered"]*100,1) if r["delivered"] else 0
        click_rate = round(r["clicked"]/r["delivered"]*100,1) if r["delivered"] else 0
        writer.writerow([
            r["name"],r["subject"],r["audience"],r["campaign_type"],r["status"],
            r["delivered"],r["opened"],r["clicked"],open_rate,click_rate
        ])

    memory = io.BytesIO(output.getvalue().encode("utf-8"))
    memory.seek(0)
    return send_file(memory,
                     mimetype="text/csv",
                     as_attachment=True,
                     download_name="campaign_report.csv")
